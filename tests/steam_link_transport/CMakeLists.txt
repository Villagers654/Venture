cmake_minimum_required(VERSION 3.28)
project(VentureSteamLinkTransportTests LANGUAGES CXX)
if(NOT WIN32)
  message(FATAL_ERROR "These tests exercise the real Windows named-pipe client")
endif()
set(CMAKE_CXX_STANDARD 23)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
get_filename_component(venture_root "${CMAKE_CURRENT_LIST_DIR}/../.." ABSOLUTE)
include(FetchContent)
FetchContent_Declare(nlohmann_json
  URL https://github.com/nlohmann/json/archive/55f93686c01528224f448c19128836e7df245f72.tar.gz
  URL_HASH SHA256=67f4cdd9ca930c9c1e130af4a437c7fc98fab77a2846fc2d2a14b4943831f8ef)
FetchContent_MakeAvailable(nlohmann_json)
add_executable(venture_pipe_fixture server.cpp)
set_target_properties(venture_pipe_fixture PROPERTIES OUTPUT_NAME vrserver)
add_executable(venture_steam_link_transport_tests client.cpp
  "${venture_root}/src/runtime/platform/win32/steam_link.cpp")
foreach(target IN ITEMS venture_pipe_fixture venture_steam_link_transport_tests)
  target_include_directories(${target} PRIVATE "${venture_root}/src" "${venture_root}/src/runtime")
  target_compile_definitions(${target} PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN UNICODE _UNICODE _WIN32_WINNT=0x0A00)
  if(MINGW)
    target_link_options(${target} PRIVATE -static)
  endif()
endforeach()
target_link_libraries(venture_steam_link_transport_tests PRIVATE nlohmann_json::nlohmann_json bcrypt)
enable_testing()
add_executable(venture_passthrough_selection_tests ../passthrough/backend_selection.cpp
  "${venture_root}/src/runtime/xr/steam_link.cpp")
target_include_directories(venture_passthrough_selection_tests PRIVATE
  "${venture_root}/src" "${venture_root}/src/runtime")
if(MINGW)
  target_link_options(venture_passthrough_selection_tests PRIVATE -static)
endif()
add_test(NAME passthrough_backend_selection COMMAND venture_passthrough_selection_tests)
add_test(NAME steam_link_real_windows_transport COMMAND venture_steam_link_transport_tests "$<TARGET_FILE:venture_pipe_fixture>")
set_tests_properties(steam_link_real_windows_transport PROPERTIES TIMEOUT 90)
